    double t2 = t * t;
    double t3 = t2 * t;
    double t4 = t3 * t;
    double t5 = t4 * t;
    double t6 = t5 * t;
    double t7 = t6 * t;
    var x = (1.0 / 5.0) * (t5 - 36 * t3 + 260 * t) / 10;
    var y = (1.0 / 2.0) * (t4 - 24 * t2) / 10;
    var z = (1.0 / 100.0) * (t7 - 31 * t5 + 164 * t3 - 560 * t) / 10;
